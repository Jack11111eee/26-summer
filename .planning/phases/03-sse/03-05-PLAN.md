## start 端点（POST /sessions/{session_id}/start——create_session :79-117 状态转换三步先例）
def start_session(session_id: str, user) -> dict:
    conn = get_conn()
    s = load_owned_session(conn, session_id, user)
    if s["status"] != "in_progress": 409 SESSION_NOT_IN_PROGRESS（既有护栏复用）
    if s["phase"] == "ACTIVE": 409 SESSION_ALREADY_ACTIVE（WR-01 三态——幂等语义：重复 start 不重复开区间）
    # 三动作同事务（§13.1 快照与事件同事务——Pitfall 12「未确认先计时」的正面锚）：
    UPDATE assessment_session SET phase='ACTIVE' WHERE session_id=?
    open_interval(conn, session_id, 'active')   # 第一个 active 区间（§15「确认开始且首题激活起算」——首题激活在 get_session 派发，区间起算此处，两者注释 §15 原文——设计裁量：区间早于首题激活 <=1 请求间隔，可接受（/start 后立即 GET 取首题，同一 UI 动作）
    append_event(..., "SESSION_STARTED", from_state="PENDING_START", to_state="ACTIVE",
                 actor_type="candidate", actor_id=user["user_id"])
    conn.commit()
    return {"session_id": session_id, "phase": "ACTIVE", "started": True}

## pause/resume 端点（POST /sessions/{id}/pause + /resume——submit_feedback :609-636 带护栏 POST 先例）
pause：
    load_owned_session → status in_progress 409 护栏（既有）→ 已有 open paused 区间 → 409 SESSION_ALREADY_PAUSED
    → close_open_interval（当前 active 闭合）→ open_interval(session_id, 'paused', reason='candidate_request')
    → append SESSION_PAUSE_REQUESTED（actor candidate）+ SESSION_PAUSED（from 'ACTIVE' to 'PAUSED'——phase 列承载；状态语义事件）——两事件同事务 conn.commit()
    → return {"session_id", "phase": "PAUSED", "paused": True}
resume：
    → 无 open paused 区间 → 409 SESSION_NOT_PAUSED
    → close_open_interval（闭合 paused）→ open_interval(session_id, 'active', reason='resumed')  # 恢复计时
    → append SESSION_RESUMED（from 'PAUSED' to 'ACTIVE'）同事务 commit
    → return {..., "phase": "ACTIVE", "resumed": True}
